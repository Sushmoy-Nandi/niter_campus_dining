const CONFIG = {
  // Replace with your actual live deployed URL (e.g., https://your-app.vercel.app)
  API_URL: 'http://localhost:3000',
  // Replace with your NEXT_PUBLIC_MASTER_SHEET_SECRET
  SECRET_KEY: 'YOUR_SECRET_KEY_HERE'
};

// Convert a 1-based column index to its A1 letter(s): 1 -> "A", 27 -> "AA", 91 -> "CM".
// Used so every formula can reference dynamically-positioned columns by letter.
function colToA1(col) {
  let s = '';
  let n = col;
  while (n > 0) {
    const rem = (n - 1) % 26;
    s = String.fromCharCode(65 + rem) + s;
    n = Math.floor((n - 1) / 26);
  }
  return s;
}

function onOpen() {
  const ui = SpreadsheetApp.getUi();
  ui.createMenu('Campus Dining Sync')
    .addItem('Sync Live Master Sheet', 'syncLiveMasterSheet')
    .addToUi();
}

function syncLiveMasterSheet() {
  const ui = SpreadsheetApp.getUi();
  try {
    const response = UrlFetchApp.fetch(`${CONFIG.API_URL}/api/admin/live-json?secret=${CONFIG.SECRET_KEY}`);
    const data = JSON.parse(response.getContentText());
    buildSheetWithData(data);
    ui.alert('Success', `Live data for "${data.period.title}" has been successfully synced and formatted!`, ui.ButtonSet.OK);
  } catch (error) {
    ui.alert('Error', `Failed to sync live data: ${error.message}`, ui.ButtonSet.OK);
  }
}

function buildSheetWithData(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(data.period.title);
  
  if (!sheet) {
    sheet = ss.insertSheet(data.period.title);
  } else {
    sheet.clear();
  }
  
  // --- DYNAMIC LAYOUT ---
  // Everything on the right side of the sheet is positioned relative to the real
  // number of days in this period (data.days.length), so a 28-, 30-, 31- or 35-day
  // period all stay perfectly aligned. Nothing here assumes a fixed 31 or 35 days.
  const numDays = data.days.length;             // e.g. 30
  const DATE_START_COL = 17;                    // Col Q: first day's Lunch column (shifted by 1 for Refund column)
  const totalMealsCol = DATE_START_COL + numDays * 2;   // right after the last day's L/D pair
  const blankCol = totalMealsCol + 1;
  const bazDateCol = totalMealsCol + 2;         // Bazaar block starts one blank col after Total Meals
  const bazNameCol = bazDateCol + 1;
  const bazDetailsCol = bazDateCol + 2;
  const bazCostCol = bazDateCol + 3;
  const lastCol = bazCostCol;

  // A1 letters for the columns formulas need to reference dynamically.
  const totalMealsA1 = colToA1(totalMealsCol);
  const dateFirstA1 = colToA1(DATE_START_COL);
  const dateLastA1 = colToA1(totalMealsCol - 1);
  const bazCostA1 = colToA1(bazCostCol);

  // Set basic dimensions
  const requiredRows = Math.max(305, data.students.length + 10);
  if (sheet.getMaxRows() < requiredRows) sheet.insertRowsAfter(sheet.getMaxRows(), requiredRows - sheet.getMaxRows());
  if (sheet.getMaxColumns() < lastCol) sheet.insertColumnsAfter(sheet.getMaxColumns(), lastCol - sheet.getMaxColumns());
  
  // Colors
  const colorA_G = '#c9daf8';
  const colorI_K = '#d9d2e9';
  const colorO_CF = '#9fc5e8';
  const colorTotalMeals = '#6aa84f';
  
  // Custom colorful deposits
  const colorDep1 = '#ffe599'; // Pastel Yellow
  const colorDep2 = '#f9cb9c'; // Pastel Orange
  const colorDep3 = '#b6d7a8'; // Pastel Green
  
  // Freeze panes up to Name (Col B)
  sheet.setFrozenRows(2);
  sheet.setFrozenColumns(2);
  
  // --- COLUMN WIDTHS ---
  sheet.setColumnWidth(1, 100); // A: ID
  sheet.setColumnWidth(2, 160); // B: Name
  sheet.setColumnWidth(3, 110); // C: Deposit 1
  sheet.setColumnWidth(4, 110); // D: Deposit 2
  sheet.setColumnWidth(5, 110); // E: Deposit 3
  sheet.setColumnWidth(6, 115); // F: Total Deposit
  sheet.setColumnWidth(7, 35);  // G: Blank
  sheet.setColumnWidth(8, 95);  // H: Deposit
  sheet.setColumnWidth(9, 95);  // I: Refund
  sheet.setColumnWidth(10, 95); // J: Cost
  sheet.setColumnWidth(11, 95); // K: On-Hand
  sheet.setColumnWidth(12, 35); // L: Blank
  sheet.setColumnWidth(13, 140);// M: Meal Rate Summary Block
  sheet.setColumnWidth(14, 35); // N: Blank
  sheet.setColumnWidth(15, 100);// O: ID
  sheet.setColumnWidth(16, 170);// P: Name

  // Date columns: P onwards, 2 columns (L/D) per day, dynamically allocated.
  let cIndex = DATE_START_COL;
  for (let i = 0; i < numDays; i++) {
    sheet.setColumnWidth(cIndex, 32);
    sheet.setColumnWidth(cIndex + 1, 32);
    cIndex += 2;
  }

  sheet.setColumnWidth(totalMealsCol, 75); // Total Meals
  sheet.setColumnWidth(blankCol, 35);       // Blank
  sheet.setColumnWidth(bazDateCol, 110);    // Date
  sheet.setColumnWidth(bazNameCol, 160);    // Name
  sheet.setColumnWidth(bazDetailsCol, 300); // Bazaar Details
  sheet.setColumnWidth(bazCostCol, 110);    // Cost
  
  // --- ROW 1 & 2 HEADERS ---
  sheet.setRowHeight(1, 40);
  sheet.setRowHeight(2, 25);
  
  const setHeader = (col, val1, val2, color, merge = true) => {
    const range1 = sheet.getRange(1, col);
    range1.setValue(val1).setFontWeight("bold").setBackground(color)
      .setHorizontalAlignment("center").setVerticalAlignment("middle")
      .setWrap(true).setBorder(true, true, true, true, false, false);
    
    if (merge) {
      sheet.getRange(1, col, 2, 1).merge();
    } else {
      const range2 = sheet.getRange(2, col);
      range2.setValue(val2).setFontWeight("bold").setBackground(color)
        .setHorizontalAlignment("center").setVerticalAlignment("middle")
        .setWrap(true).setBorder(true, true, true, true, false, false);
    }
  };
  
  const lastRow = data.students.length + 2;
  const totalRowIdx = lastRow + 1;
  
  // Left Section Headers
  setHeader(1, 'ID', '', colorA_G);
  setHeader(2, 'Name', '', colorA_G);
  
  // Deposit Header Formulas with explicitly formatted totals
  sheet.getRange(1, 3).setFormula(`="Deposit 1 "&CHAR(10)&TEXT(SUM(C${totalRowIdx}+0), "0.00")`);
  sheet.getRange(1, 3, 2, 1).merge().setFontWeight("bold").setBackground(colorDep1)
    .setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, false, false);
    
  sheet.getRange(1, 4).setFormula(`="Deposit 2 "&CHAR(10)&TEXT(SUM(D${totalRowIdx}+0), "0.00")`);
  sheet.getRange(1, 4, 2, 1).merge().setFontWeight("bold").setBackground(colorDep2)
    .setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, false, false);
    
  sheet.getRange(1, 5).setFormula(`="Deposit 3 "&CHAR(10)&TEXT(SUM(E${totalRowIdx}+0), "0.00")`);
  sheet.getRange(1, 5, 2, 1).merge().setFontWeight("bold").setBackground(colorDep3)
    .setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, false, false);
    
  sheet.getRange(1, 6).setFormula(`="Total Deposit "&CHAR(10)&TEXT(SUM(F${totalRowIdx}+0), "0.00")`);
  sheet.getRange(1, 6, 2, 1).merge().setFontWeight("bold").setBackground(colorA_G)
    .setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, false, false);
    
  setHeader(8, 'Deposit', '', colorI_K);
  setHeader(9, 'Refund', '', colorI_K);
  setHeader(10, 'Cost', '', colorI_K);
  setHeader(11, 'On-Hand', '', colorI_K);
  
  // Right Section Headers
  setHeader(15, 'ID', '', colorO_CF);
  setHeader(16, 'Name', '', colorO_CF);
  
  let dateCol = DATE_START_COL; // Col P
  for (let i = 0; i < numDays; i++) {
    const dStr = data.days[i];
    const range = sheet.getRange(1, dateCol);
    if (dStr) {
      range.setValue(new Date(dStr)).setNumberFormat('d-MMM');
    }
    range.setFontWeight("bold").setBackground(colorO_CF)
      .setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, false, false);
    sheet.getRange(1, dateCol, 1, 2).merge();
    sheet.getRange(1, dateCol + 1).setBorder(true, true, true, true, false, false);
    sheet.getRange(2, dateCol).setValue('L').setFontWeight("normal").setBackground(colorO_CF)
      .setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, false, false);
    sheet.getRange(2, dateCol + 1).setValue('D').setFontWeight("normal").setBackground(colorO_CF)
      .setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, false, false);
    dateCol += 2;
  }

  setHeader(totalMealsCol, 'Total Meals', '', colorO_CF);
  setHeader(bazDateCol, 'Date', '', colorO_CF);
  setHeader(bazNameCol, 'Name', '', colorO_CF);
  setHeader(bazDetailsCol, 'Bazaar Details', '', colorO_CF);
  setHeader(bazCostCol, 'Cost', '', colorO_CF);
  
  // --- STUDENT ROWS DATA ---
  if (data.students.length > 0) {
    let r = 3;
    const values = [];
    
    for (const student of data.students) {
      const row = new Array(lastCol).fill('');
      // Individual deposits come as an array (Deposit 1..3); show each in its own
      // column so admins see the real breakdown instead of the lump total.
      const deps = student.deposits || [student.deposit];
      row[0] = student.diningId;       // A: ID
      row[1] = student.name;           // B: Name
      row[2] = deps[0] != null ? deps[0] : 0; // C: Deposit 1
      row[3] = deps[1] != null ? deps[1] : 0; // D: Deposit 2
      row[4] = deps[2] != null ? deps[2] : 0; // E: Deposit 3
      row[5] = `=SUM(C${r}:E${r})`;    // F: Total Deposit
      row[7] = `=SUM(F${r}+0)`;        // H: Deposit
      row[8] = student.refund || 0;    // I: Refund
      row[9] = `=${totalMealsA1}${r}*$M$4`; // J: Cost = meals * rate ($M$4)
      row[10] = `=H${r}-I${r}-J${r}`; // K: On-Hand = Deposit - Refund - Cost
      row[14] = student.diningId;      // O: ID
      row[15] = student.name;          // P: Name

      let mc = DATE_START_COL - 1; // 0-based index of Col P
      for (let i = 0; i < numDays; i++) {
        const mealData = student.meals[i];
        if (mealData) {
          row[mc] = mealData.l;
          row[mc+1] = mealData.d;
        } else {
          row[mc] = '';
          row[mc+1] = '';
        }
        mc += 2;
      }
      // Total Meals = sum of every L/D cell for this student (dynamic range).
      row[totalMealsCol - 1] = `=SUM(${dateFirstA1}${r}:${dateLastA1}${r})`;
      values.push(row);
      r++;
    }

    const dataRange = sheet.getRange(3, 1, data.students.length, lastCol);
    dataRange.setValues(values);
    dataRange.setHorizontalAlignment("center");

    // Set specific borders
    sheet.getRange(3, 1, data.students.length, 6).setBorder(true, true, true, true, false, false);
    sheet.getRange(3, 8, data.students.length, 4).setBorder(true, true, true, true, false, false);
    sheet.getRange(3, 15, data.students.length, (totalMealsCol - 15 + 1)).setBorder(true, true, true, true, false, false);
    sheet.getRange(3, totalMealsCol, data.students.length, 1).setBackground(colorTotalMeals).setNumberFormat('0');

    // Strict Number Formats to avoid Date Auto-formatting
    sheet.getRange(3, 3, data.students.length, 4).setNumberFormat('0.00'); // C, D, E, F
    sheet.getRange(3, 8, data.students.length, 4).setNumberFormat('0.00'); // H, I, J, K
  }
  
  const lastBazRow = Math.max(3, 2 + data.bazaars.length);

  // --- MEAL RATE SUMMARY BLOCK (Column M, starts at Row 3) ---
  sheet.getRange('M3').setValue('Current Meal Rate').setFontWeight("bold").setHorizontalAlignment("center");
  sheet.getRange('M4').setFormula('=IFERROR(M11/M7, 0)').setHorizontalAlignment("center").setFontWeight("bold").setNumberFormat('0.00');
  sheet.getRange('M6').setValue('Total Meal Count').setFontWeight("bold").setHorizontalAlignment("center");
  sheet.getRange('M7').setFormula(`=SUM(${totalMealsA1}3:${totalMealsA1}${lastRow})`).setNumberFormat('0').setHorizontalAlignment("center");
  sheet.getRange('M10').setValue('Total Cost').setFontWeight("bold").setHorizontalAlignment("center");
  sheet.getRange('M11').setFormula(`=SUM(${bazCostA1}3:${bazCostA1}${lastBazRow})`).setNumberFormat('0.00').setHorizontalAlignment("center");
  sheet.getRange('M13').setValue('On-Hand').setFontWeight("bold").setHorizontalAlignment("center");
  sheet.getRange('M14').setFormula(`=SUM(K3:K${lastRow})`).setNumberFormat('0.00').setHorizontalAlignment("center");
  
  sheet.getRange('M3:M4').setBorder(true, true, true, true, false, false);
  sheet.getRange('M6:M7').setBorder(true, true, true, true, false, false);
  sheet.getRange('M10:M11').setBorder(true, true, true, true, false, false);
  sheet.getRange('M13:M14').setBorder(true, true, true, true, false, false);
  
  // --- TOTALS ROW ---
  const tr = totalRowIdx;
  sheet.getRange(tr, 3).setFormula(`=SUM(C3:C${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00');
  sheet.getRange(tr, 4).setFormula(`=SUM(D3:D${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00');
  sheet.getRange(tr, 5).setFormula(`=SUM(E3:E${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00');
  sheet.getRange(tr, 6).setFormula(`=SUM(F3:F${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00').setBorder(true, true, true, true, false, false);
  
  sheet.getRange(tr, 8).setFormula(`=SUM(H3:H${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00').setBorder(true, true, true, true, false, false);
  sheet.getRange(tr, 9).setFormula(`=SUM(I3:I${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00').setBorder(true, true, true, true, false, false);
  sheet.getRange(tr, 10).setFormula(`=SUM(J3:J${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00').setBorder(true, true, true, true, false, false);
  sheet.getRange(tr, 11).setFormula(`=SUM(K3:K${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00').setBorder(true, true, true, true, false, false);

  // Totals for every day column (Q .. last day), summing each column down.
  for (let c = DATE_START_COL; c < totalMealsCol; c++) {
    const ltr = colToA1(c);
    sheet.getRange(tr, c).setFormula(`=SUM(${ltr}3:${ltr}${lastRow})`).setFontWeight("bold").setHorizontalAlignment("center").setBorder(true, true, true, true, false, false);
  }

  // Grand total of every student's Total Meals (sum the Total Meals column down).
  sheet.getRange(tr, totalMealsCol).setFormula(`=SUM(${totalMealsA1}3:${totalMealsA1}${lastRow})`).setBackground(colorTotalMeals).setFontWeight("bold").setHorizontalAlignment("center").setBorder(true, true, true, true, false, false);
  // Grand total of all bazaar costs (summing up to lastBazRow).
  sheet.getRange(tr, bazCostCol).setFormula(`=SUM(${bazCostA1}3:${bazCostA1}${lastBazRow})`).setFontWeight("bold").setHorizontalAlignment("center").setNumberFormat('0.00').setBorder(true, true, true, true, false, false);

  // --- BAZAAR SECTION DATA ---
  const bazaarRowsCount = Math.max(data.bazaars.length, 100);
  sheet.getRange(3, bazCostCol, bazaarRowsCount, 1).setNumberFormat('0.00');

  let bRow = 3;
  for (const b of data.bazaars) {
    sheet.getRange(bRow, bazDateCol).setValue(new Date(b.date)).setNumberFormat('dd-MMM-yyyy').setBorder(true, true, true, true, false, false);
    sheet.getRange(bRow, bazNameCol).setValue(b.name || 'Bazaar').setBorder(true, true, true, true, false, false);
    sheet.getRange(bRow, bazDetailsCol).setValue(b.details || '').setBorder(true, true, true, true, false, false);
    sheet.getRange(bRow, bazCostCol).setValue(Number(b.amount) || 0).setNumberFormat('0.00').setBorder(true, true, true, true, false, false);
    bRow++;
  }
}

function syncLiveMasterSheetBackground() {
  try {
    const response = UrlFetchApp.fetch(`${CONFIG.API_URL}/api/admin/live-json?secret=${CONFIG.SECRET_KEY}`);
    const data = JSON.parse(response.getContentText());
    buildSheetWithData(data);
  } catch (error) {
    console.error("Background sync error: " + error.message);
  }
}

function doPost(e) {
  // This runs the sync whenever the website pings it, without crashing on getUi()
  syncLiveMasterSheetBackground();
  return ContentService.createTextOutput("Sync triggered successfully!");
}
